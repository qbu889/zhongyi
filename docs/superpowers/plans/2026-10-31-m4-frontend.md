# M4 前端实施计划（对话页 + 四大库 + 溯源抽屉 + 问诊引导）

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付可用的 Web 前端：流式对话与可见出处、四大库浏览、溯源抽屉、问诊卡，并通过 PRD §6.7 交互验收。

**Architecture:** Next.js App Router；四大库列表/详情用 Server Component 直取 FastAPI，对话页为 Client Component 走 SSE；浏览器统一经 `/api/*` rewrite 代理后端。后端只做最小补口（会话列表/删除、SSE 进度与错误事件、列表多值筛选、E2E 引导脚本），不改 `Answer` 结构。

**Tech Stack:** Next.js 16.4.3（App Router）、React 19.3.1、TypeScript 5.9.3、Tailwind CSS 4.3.3、Radix UI、Vitest 5.0.5 + RTL + MSW、Playwright 1.64.1 + axe-core。

**Spec:** `docs/superpowers/specs/2026-10-31-m4-frontend-design.md`

- **剂量纪律**：克数只来自 `dose_g`；`approx` 显示 `~{g}g` + 橙色；`count_only / unknown_unit / absent` 克数列显示 `—`；不得把古制数量渲染成精确克数。

- **E2E 数据库**：库名必须含 `e2e`（默认 `zhongyi_e2e`）；bootstrap 对非 e2e 库直接拒绝。
- **M4 不做**：FR-14 原文检索（按钮置灰）、会话重命名、反馈落库、框架页/管理页真实内容、compose 前端、鉴权、生产建表、真实 LLM。

```python
# backend/app/dev/bootstrap.py
"""E2E 专用建库+重建脚本；生产建表路径仍留 M5。"""

from sqlalchemy import create_engine, text
from sqlalchemy.orm import Session

from app.admin.reindex import run_reindex
from app.db.models import Base
from app.db.schema import ensure_search_indexes


def is_e2e_url(url: str) -> bool:
    name = url.rsplit("/", 1)[-1].split("?", 1)[0]
    return bool(name) and "e2e" in name.lower()


def bootstrap(url: str) -> dict:
    if not is_e2e_url(url):
        raise RuntimeError(f"拒绝在非 e2e 库上重建：{url}")
    engine = create_engine(url, future=True)
    try:
        with engine.begin() as conn:
            conn.execute(text("CREATE EXTENSION IF NOT EXISTS vector"))
        Base.metadata.drop_all(engine)
        Base.metadata.create_all(engine)
        with engine.begin() as conn:
            ensure_search_indexes(conn)
        with Session(engine) as session:
            report = run_reindex(session)
            session.commit()
        return {"version_id": report.version_id, "chunks": report.chunks}
    finally:
        engine.dispose()
```

```python

def _session_title(session: Session, session_id: int) -> str:
    content = session.scalar(
        select(ChatMessage.content)
        .where(ChatMessage.session_id == session_id, ChatMessage.role == "user")
        .order_by(ChatMessage.id).limit(1))
    text = (content or {}).get("text", "")
    if not text:
        return "新会话"
    return text[:30] + ("…" if len(text) > 30 else "")

@router.get("/sessions", response_model=SessionListResponse)
def list_sessions(
    limit: int = Query(default=50, ge=1, le=100),
    offset: int = Query(default=0, ge=0),
    session: Session = Depends(get_session),
):
    total = session.scalar(select(func.count()).select_from(ChatSession)) or 0
    rows = session.execute(
        select(
            ChatSession.id,
            func.max(ChatMessage.id).label("last_msg_id"),
            func.max(ChatMessage.created_at).label("last_at"),
            func.count(ChatMessage.id).label("cnt"),
        )
        .outerjoin(ChatMessage, ChatMessage.session_id == ChatSession.id)
        .group_by(ChatSession.id)
        .order_by(func.coalesce(func.max(ChatMessage.id), 0).desc(),
                  ChatSession.id.desc())
        .limit(limit).offset(offset)).all()

    items = []
    for sid, _last_msg_id, last_at, cnt in rows:
        chat = session.get(ChatSession, sid)
        items.append(SessionItem(
            id=sid, title=_session_title(session, sid),
            last_activity_at=last_at or chat.created_at, message_count=cnt))
    return SessionListResponse(total=total, limit=limit, offset=offset, items=items)


@router.delete("/{session_id}", status_code=204)
def delete_session(session_id: int, session: Session = Depends(get_session)):
    chat = session.get(ChatSession, session_id)
    if chat is None:
        raise HTTPException(404, detail={"code": "session_not_found", "id": session_id})
    session.execute(update(QaLog).where(QaLog.session_id == session_id)
                    .values(session_id=None))
    session.execute(delete(ChatMessage).where(ChatMessage.session_id == session_id))
    session.delete(chat)
    session.commit()
    return Response(status_code=204)
```

```python
from collections import Counter


def _stream_error(exc: Exception) -> dict:
    if isinstance(exc, retriever.IndexNotReady):
        return {"code": "index_not_ready", "message": "索引尚未构建，请先调用 /api/admin/reindex"}
    if isinstance(exc, retriever.EmbeddingMismatch):
        return {"code": "embedding_mismatch", "message": str(exc)}
    if isinstance(exc, ValueError):
        return {"code": "bad_request", "message": str(exc)}
    return {"code": "internal_error", "message": "生成失败"}


# 替换原 stream_events
def stream_events(session, question: str, *, session_id: int | None = None,
                  llm=None, chunk_size: int = 20):
    try:
        resp = answer_question(session, question, session_id=session_id, llm=llm)
        # 先提交：客户端中断时该轮仍可在历史中回放
        session.commit()
        answer = resp.answer
        yield (f"event: meta\ndata: "
               f"{json.dumps({'session_id': resp.session_id, 'intent': answer.intent, 'status': answer.status}, ensure_ascii=False)}\n\n")
        counts = Counter(c.ref_type for c in answer.citations)
        yield (f"event: progress\ndata: "
               f"{json.dumps({'total': len(answer.citations), 'by_type': dict(counts)}, ensure_ascii=False)}\n\n")
        for citation in answer.citations:
            yield f"event: citation\ndata: {citation.model_dump_json()}\n\n"
        for start in range(0, len(answer.prose), chunk_size):
            yield (f"event: token\ndata: "
                   f"{json.dumps({'text': answer.prose[start:start + chunk_size]}, ensure_ascii=False)}\n\n")
        yield f"event: done\ndata: {resp.model_dump_json()}\n\n"
    except Exception as exc:  # GeneratorExit 不继承 Exception，中断不会误报
        payload = _stream_error(exc)
        yield f"event: error\ndata: {json.dumps(payload, ensure_ascii=False)}\n\n"
```

```ts
export async function parseApiError(res: Response): Promise<ApiError> {
  let code = "http_error";
  let message = `HTTP ${res.status}`;
  try {
    const body = (await res.json()) as { detail?: { code?: string; message?: string } };
    code = body.detail?.code ?? code;
    message = body.detail?.message ?? message;
  } catch {
    // 非 JSON 错误体保持默认文案
  }
  return new ApiError(code, message, res.status);
}

export async function serverGet<T>(path: string): Promise<T> {

  if (!res.ok) throw await parseApiError(res);

}
```

```ts
// frontend/lib/sse.ts
import { ApiError, parseApiError } from "./api";
import type { ChatResponse, Citation, MetaEvent, Progress } from "./types";

export interface SseHandlers {
  onMeta?: (meta: MetaEvent) => void;
  onProgress?: (progress: Progress) => void;
  onCitation?: (citation: Citation) => void;
  onToken?: (text: string) => void;
  onDone?: (response: ChatResponse) => void;
  onError?: (error: { code: string; message: string }) => void;
}

export function parseSseFrame(frame: string): { event: string; data: unknown } | null {
  let event = "message";
  const data: string[] = [];
  for (const raw of frame.split("\n")) {
    const line = raw.trimEnd();
    if (!line || line.startsWith(":")) continue;
    if (line.startsWith("event:")) event = line.slice(6).trim();
    else if (line.startsWith("data:")) data.push(line.slice(5).replace(/^ /, ""));
  }
  if (!data.length) return null;
  const text = data.join("\n");
  let parsed: unknown = text;
  try {
    parsed = JSON.parse(text);
  } catch {
    // 非 JSON data 保留原字符串
  }
  return { event, data: parsed };
}

export async function streamChat(
  question: string,
  sessionId: number | null,
  handlers: SseHandlers,
  signal?: AbortSignal,
): Promise<void> {
  const res = await fetch("/api/chat/stream", {
    method: "POST",
    headers: { "Content-Type": "application/json", Accept: "text/event-stream" },
    body: JSON.stringify({ question, session_id: sessionId }),
    signal,
  });
  if (!res.ok) throw await parseApiError(res);
  if (!res.body) throw new ApiError("no_body", "流式响应不可读", 0);

  const reader = res.body.getReader();
  const decoder = new TextDecoder();
  let buffer = "";
  for (;;) {
    const { value, done } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });
    let index = buffer.indexOf("\n\n");
    while (index >= 0) {
      const frame = buffer.slice(0, index);
      buffer = buffer.slice(index + 2);
      const parsed = parseSseFrame(frame);
      if (parsed) {
        switch (parsed.event) {
          case "meta": handlers.onMeta?.(parsed.data as MetaEvent); break;
          case "progress": handlers.onProgress?.(parsed.data as Progress); break;
          case "citation": handlers.onCitation?.(parsed.data as Citation); break;
          case "token": handlers.onToken?.((parsed.data as { text: string }).text); break;
          case "done": handlers.onDone?.(parsed.data as ChatResponse); break;
          case "error": handlers.onError?.(parsed.data as { code: string; message: string }); break;
          default: break;
        }
      }
      index = buffer.indexOf("\n\n");
    }
  }
}
```

```ts
import type { Citation, Section } from "./types";

export function citationKey(citation: Citation): string {
  return `${citation.ref_type}:${citation.ref_id}`;
}

export function numberCitations(citations: Citation[]): Map<string, number> {
  const numbers = new Map<string, number>();
  for (const citation of citations) {
    const key = citationKey(citation);
    if (!numbers.has(key)) numbers.set(key, numbers.size + 1);
  }
  return numbers;
}

export function numbersForLine(
  line: string,
  citations: Citation[],
  numbers: Map<string, number>,
): number[] {
  const found = new Set<number>();
  for (const citation of citations) {
    if (citation.title && line.includes(citation.title)) {
      const number = numbers.get(citationKey(citation));
      if (number) found.add(number);
    }
  }
  return [...found].sort((a, b) => a - b);
}

export function unmatchedCitations(sections: Section[], citations: Citation[]): Citation[] {
  const seen = new Set<string>();
  const unmatched: Citation[] = [];
  for (const citation of citations) {
    const key = citationKey(citation);
    if (seen.has(key)) continue;
    seen.add(key);
    const hit = sections.some((section) =>
      section.lines.some((line) => citation.title && line.includes(citation.title)));
    if (!hit) unmatched.push(citation);
  }
  return unmatched;
}
```

```ts
export type FilterValue = string | string[] | number | null | undefined;

export function buildQuery(values: Record<string, FilterValue>): string {
  const params = new URLSearchParams();
  for (const [key, value] of Object.entries(values)) {
    if (value === null || value === undefined || value === "") continue;
    if (Array.isArray(value)) {
      for (const item of value) if (item) params.append(key, item);
    } else {
      params.set(key, String(value));
    }
  }
  const query = params.toString();
  return query ? `?${query}` : "";
}

export function readList(params: URLSearchParams, key: string): string[] {
  return params.getAll(key).filter(Boolean);
}

export function pageFrom(params: URLSearchParams): number {
  const value = Number(params.get("page") ?? "1");
  return Number.isFinite(value) && value >= 1 ? Math.floor(value) : 1;
}

export function pageHref(path: string, params: URLSearchParams, page: number): string {
  const next = new URLSearchParams(params);
  if (page <= 1) next.delete("page");
  else next.set("page", String(page));
  const query = next.toString();
  return query ? `${path}?${query}` : path;
}
```

```tsx
  const index = snippet ? text.indexOf(snippet) : -1;
  if (index < 0) return text;
  return (
    <>
      {text.slice(0, index)}
      <mark className="bg-cinnabar/15">{snippet}</mark>
      {text.slice(index + snippet.length)}
    </>
  );

export function EvidenceDrawer({
  citation,
  onClose,
}: {
  citation: Citation | null;
  onClose: () => void;
}) {
  const [state, setState] = useState<State>({ kind: "loading" });

    if (!citation) return;
    let cancelled = false;
    setState({ kind: "loading" });
```

```tsx
// frontend/components/cards/useCards.ts
"use client";

import { useEffect, useState } from "react";

import { apiGet } from "@/lib/api";
import { citationKey } from "@/lib/citations";
import type {
  AcupointDetail, CaseDetail, Citation, FormulaDetail, HerbDetail,
} from "@/lib/types";

type CardState =
  | { kind: "formula"; data: FormulaDetail }
  | { kind: "herb"; data: HerbDetail }
  | { kind: "acupoint"; data: AcupointDetail }
  | { kind: "case"; data: CaseDetail }
  | { kind: string; error: true };

function load(citation: Citation): Promise<CardState> {
  switch (citation.ref_type) {
    case "formula":
      return apiGet<FormulaDetail>(`/api/formulas/${citation.ref_id}`)
        .then((data) => ({ kind: "formula", data }));
    case "herb":
      return apiGet<HerbDetail>(`/api/herbs/${citation.ref_id}`)
        .then((data) => ({ kind: "herb", data }));
    case "acupoint":
      return apiGet<AcupointDetail>(`/api/acupoints/${citation.ref_id}`)
        .then((data) => ({ kind: "acupoint", data }));
    case "case":
      return apiGet<CaseDetail>(`/api/cases/${citation.ref_id}`)
        .then((data) => ({ kind: "case", data }));
    default:
      return Promise.resolve({ kind: citation.ref_type, error: true });
  }
}

export function useCards(citations: Citation[]) {
  const [cards, setCards] = useState<Record<string, CardState>>({});
  const [nonce, setNonce] = useState(0);

  useEffect(() => {
    let cancelled = false;
    const seen = new Set<string>();
    const targets = citations.filter((citation) => {
      if (citation.ref_type === "evidence" || citation.ref_type === "framework") return false;
      const key = citationKey(citation);
      if (seen.has(key)) return false;
      seen.add(key);
      return true;
    });
    Promise.all(targets.map(async (citation) => {
      const key = citationKey(citation);
      try {
        return [key, await load(citation)] as const;
      } catch {
        return [key, { kind: citation.ref_type, error: true }] as const;
      }
    })).then((entries) => {
      if (!cancelled) setCards(Object.fromEntries(entries));
    });
    return () => { cancelled = true; };
  }, [citations, nonce]);

  return { cards, reload: () => setNonce((value) => value + 1) };
}
```

```tsx
import { AcupointCard } from "./AcupointCard";
import { CaseCard } from "./CaseCard";
import { FormulaCard } from "./FormulaCard";
import { HerbCard } from "./HerbCard";
import { useCards } from "./useCards";

export function Cards({ citations }: { citations: Citation[] }) {
  const { cards, reload } = useCards(citations);
  const entries = Object.entries(cards);

  if (entries.length === 0) return null;

  return (
    <div data-testid="cards" className="mt-3 space-y-2">
      {entries.map(([key, card]) => {
        if ("error" in card) {
          return (
            <div key={key} className="rounded border border-line p-2 text-sm text-ink/60">
              卡片加载失败
              <button type="button" className="ml-2 text-cinnabar underline" onClick={reload}>重试</button>
            </div>
          );
        }
        if (card.kind === "formula") return <FormulaCard key={key} formula={card.data} />;
        if (card.kind === "herb") return <HerbCard key={key} herb={card.data} />;
        if (card.kind === "acupoint") return <AcupointCard key={key} point={card.data} />;
        return <CaseCard key={key} caseItem={card.data} />;
      })}
    </div>
  );
}
```

```tsx
// frontend/components/intake/IntakeCard.tsx
"use client";

import * as Tabs from "@radix-ui/react-tabs";
import { useState } from "react";

import {
  formatIntakeSummary,
  prefillIntake,
  SEVEN_BASELINES,
  TEN_QUESTIONS,
  TRI_LABELS,
  type IntakeDraft,
  type IntakeField,
  type TriState,
} from "@/lib/intake";

const STATES: Exclude<TriState, "">[] = ["normal", "abnormal", "unknown"];

function FieldRow({ field, onChange }: {
  field: IntakeField;
  onChange: (next: IntakeField) => void;
}) {
  return (
    <div className="flex flex-wrap items-center gap-2 border-b border-line/60 py-2">
      <span className="w-24 text-sm">{field.key}</span>
      <div role="radiogroup" aria-label={field.key} className="flex gap-1">
        {STATES.map((state) => (
          <button key={state} type="button" role="radio" aria-checked={field.state === state}
            onClick={() => onChange({
              ...field,
              state,
              note: state === "abnormal" ? field.note : "",
              source: null,
            })}
            className={`rounded border px-2 py-1 text-sm ${
              field.state === state
                ? "border-cinnabar bg-cinnabar/10 text-cinnabar"
                : "border-line text-ink/70"
            }`}>
            {TRI_LABELS[state]}
          </button>
        ))}
      </div>
      {field.state === "abnormal" && (
        <input aria-label={`${field.key}补充说明`} value={field.note}
          onChange={(event) => onChange({ ...field, note: event.target.value, source: null })}
          placeholder="具体表现（选填）"
          className="w-40 rounded border border-line bg-paper px-2 py-1 text-sm" />
      )}
      {field.source === "typed" && (
        <span className="text-xs text-cinnabar">已从您的描述提取</span>
      )}
      {field.source === "history" && (
        <span className="text-xs text-ink/50">已从历史对话提取</span>
      )}
    </div>
  );
}

function FieldGroup({ title, fields, onChange }: {
  title: string;
  fields: IntakeField[];
  onChange: (index: number, next: IntakeField) => void;
}) {
  return (
    <fieldset className="mt-3">
      <legend className="text-sm font-semibold text-cinnabar">{title}</legend>
      {fields.map((field, index) => (
        <FieldRow key={field.key} field={field} onChange={(next) => onChange(index, next)} />
      ))}
    </fieldset>
  );
}

export function IntakeCard({
  followup,
  initialText,
  onSubmit,
  onSkip,
}: {
  followup: string[];
  initialText: string;
  onSubmit: (summary: string) => void;
  onSkip: () => void;
}) {
  const [draft, setDraft] = useState<IntakeDraft>(() => prefillIntake(initialText));

  const update = (kind: "baselines" | "questions") => (index: number, next: IntakeField) =>
    setDraft((prev) => ({
      ...prev,
      [kind]: prev[kind].map((field, i) => (i === index ? next : field)),
    }));

  const hasAny = [...draft.baselines, ...draft.questions].some((field) => field.state);

  return (
    <form data-testid="intake-card"
      className="mt-3 rounded border border-line bg-paper p-3"
      onSubmit={(event) => {
        event.preventDefault();
        if (hasAny) onSubmit(formatIntakeSummary(draft));
      }}>
      <h3 className="font-serif text-lg">补充问诊信息</h3>
      {followup.length > 0 && (
        <ul className="mt-1 list-disc pl-5 text-sm text-ink/70">
          {followup.map((item) => <li key={item}>{item}</li>)}
        </ul>
      )}
      <Tabs.Root defaultValue="baseline" className="mt-3">
        <Tabs.List aria-label="问诊分类" className="flex gap-2 border-b border-line">
          <Tabs.Trigger value="baseline"
            className="px-2 py-1 text-sm data-[state=active]:border-b-2 data-[state=active]:border-cinnabar">
            七条基准
          </Tabs.Trigger>
          <Tabs.Trigger value="questions"
            className="px-2 py-1 text-sm data-[state=active]:border-b-2 data-[state=active]:border-cinnabar">
            十问
          </Tabs.Trigger>
        </Tabs.List>
        <Tabs.Content value="baseline">
          <FieldGroup title="七条基准" fields={draft.baselines} onChange={update("baselines")} />
        </Tabs.Content>
        <Tabs.Content value="questions">
          <FieldGroup title="十问" fields={draft.questions} onChange={update("questions")} />
        </Tabs.Content>
      </Tabs.Root>
      <div className="mt-3 flex flex-wrap gap-2">
        <button type="submit" disabled={!hasAny}
          className="rounded bg-cinnabar px-3 py-1 text-sm text-paper disabled:opacity-50">
          提交主诉补充
        </button>
        <button type="button" onClick={onSkip}
          className="rounded border border-line px-3 py-1 text-sm text-ink/70">
          跳过，直接看回答
        </button>
      </div>
      <p className="mt-2 text-xs text-ink/60">
        只做结构化记录，不生成诊断结论；Tab 遍历、空格选择、Enter 提交。
      </p>
    </form>
  );
}
```

```ts
import type {
  Answer, AnswerStatus, ChatResponse, Citation, MetaEvent, Progress,
} from "./types";

export type TurnPhase =
  | "sending" | "intent" | "retrieving" | "streaming"
  | "done" | "refused" | "needs_more_info" | "error" | "interrupted";

export interface Turn {
  id: string;
  question: string;
  phase: TurnPhase;
  sessionId: number | null;
  messageId: number | null;
  intent: string;
  progress: Progress | null;
  citations: Citation[];
  streamText: string;
  answer: Answer | null;
  error: string | null;
  intakeSkipped: boolean;
}

export interface ChatState {
  turns: Turn[];
  sessionId: number | null;
  pending: boolean;
}

export const initialChatState: ChatState = { turns: [], sessionId: null, pending: false };

export type ChatAction =
  | { type: "send"; id: string; question: string }
  | { type: "meta"; id: string; meta: MetaEvent }
  | { type: "progress"; id: string; progress: Progress }
  | { type: "citation"; id: string; citation: Citation }
  | { type: "token"; id: string; text: string }
  | { type: "done"; id: string; response: ChatResponse }
  | { type: "error"; id: string; message: string }
  | { type: "interrupt"; id: string }
  | { type: "skip_intake"; id: string }
  | { type: "hydrate"; sessionId: number; turns: Turn[] };

function phaseFor(status: AnswerStatus): TurnPhase {
  if (status === "refused") return "refused";
  if (status === "needs_more_info") return "needs_more_info";
  return "done";
}

function patchTurn(state: ChatState, id: string, patch: (turn: Turn) => Turn): ChatState {
  return { ...state, turns: state.turns.map((turn) => (turn.id === id ? patch(turn) : turn)) };
}

export function chatReducer(state: ChatState, action: ChatAction): ChatState {
  switch (action.type) {
    case "send":
      return {
        ...state,
        sessionId: state.sessionId,
        pending: true,
        turns: [...state.turns, {
          id: action.id,
          question: action.question,
          phase: "sending",
          sessionId: state.sessionId,
          messageId: null,
          intent: "",
          progress: null,
          citations: [],
          streamText: "",
          answer: null,
          error: null,
          intakeSkipped: false,
        }],
      };
    case "meta":
      return {
        ...patchTurn(state, action.id, (turn) => ({
          ...turn, phase: "intent", intent: action.meta.intent, sessionId: action.meta.session_id,
        })),
        sessionId: action.meta.session_id,
      };
    case "progress":
      return patchTurn(state, action.id, (turn) =>
        ({ ...turn, phase: "retrieving", progress: action.progress }));
    case "citation":
      return patchTurn(state, action.id, (turn) => {
        const key = `${action.citation.ref_type}:${action.citation.ref_id}`;
        const exists = turn.citations.some(
          (item) => `${item.ref_type}:${item.ref_id}` === key);
        return exists ? turn : { ...turn, citations: [...turn.citations, action.citation] };
      });
    case "token":
      return patchTurn(state, action.id, (turn) =>
        ({ ...turn, phase: "streaming", streamText: turn.streamText + action.text }));
    case "done":
      return {
        ...patchTurn(state, action.id, (turn) => ({
          ...turn,
          phase: phaseFor(action.response.answer.status),
          sessionId: action.response.session_id,
          messageId: action.response.message_id,
          answer: action.response.answer,
          citations: action.response.answer.citations,
          streamText: "",
        })),
        sessionId: action.response.session_id,
        pending: false,
      };
    case "error":
      return {
        ...patchTurn(state, action.id, (turn) =>
          ({ ...turn, phase: "error", error: action.message })),
        pending: false,
      };
    case "interrupt":
      return {
        ...patchTurn(state, action.id, (turn) => ({ ...turn, phase: "interrupted" })),
        pending: false,
      };
    case "skip_intake":
      return patchTurn(state, action.id, (turn) => ({ ...turn, intakeSkipped: true }));
    case "hydrate":
      return { turns: action.turns, sessionId: action.sessionId, pending: false };
    default:
      return state;
  }
}
```

```ts
"use client";

import { useCallback, useRef, type Dispatch } from "react";

import { streamChat } from "@/lib/sse";

import type { ChatAction } from "@/lib/chatReducer";

export function useChatTurn(
  dispatch: Dispatch<ChatAction>,
  onSettled?: () => void,
) {
  const controllerRef = useRef<AbortController | null>(null);

  const run = useCallback(async (id: string, question: string, sessionId: number | null) => {
    let sawToken = false;
    let activeSession = sessionId;

    const attempt = async () => {
      controllerRef.current = new AbortController();
      await streamChat(question, activeSession, {
        onMeta: (meta) => {
          activeSession = meta.session_id;
          dispatch({ type: "meta", id, meta });
        },
        onProgress: (progress) => dispatch({ type: "progress", id, progress }),
        onCitation: (citation) => dispatch({ type: "citation", id, citation }),
        onToken: (text) => {
          sawToken = true;
          dispatch({ type: "token", id, text });
        },
        onDone: (response) => dispatch({ type: "done", id, response }),
        onError: (error) => dispatch({ type: "error", id, message: error.message }),
      }, controllerRef.current.signal);
    };

    try {
      try {
        await attempt();
      } catch (first) {
        if (controllerRef.current?.signal.aborted) {
          dispatch({ type: "interrupt", id });
          return;
        }
        if (sawToken) {
          dispatch({
            type: "error", id,
            message: first instanceof Error ? first.message : "生成失败",
          });
          return;
        }
        try {
          await attempt();
        } catch (second) {
          if (controllerRef.current?.signal.aborted) {
            dispatch({ type: "interrupt", id });
            return;
          }
          dispatch({
            type: "error", id,
            message: second instanceof Error ? second.message : "生成失败",
          });
        }
      }
    } finally {
      onSettled?.();
    }
  }, [dispatch, onSettled]);

  const stop = useCallback(() => controllerRef.current?.abort(), []);

  return { run, stop };
}
```

```tsx
"use client";

import { useEffect, useState } from "react";

export function Composer({
  pending,
  initialValue = "",
  onSend,
  onStop,
}: {
  pending: boolean;
  initialValue?: string;
  onSend: (text: string) => void;
  onStop: () => void;
}) {
  const [value, setValue] = useState(initialValue);

  useEffect(() => setValue(initialValue), [initialValue]);

  const submit = () => {
    const text = value.trim();
    if (!text || pending) return;
    onSend(text);
    setValue("");
  };

  return (
    <form className="sticky bottom-12 mt-4 border-t border-line bg-paper p-2"
      onSubmit={(event) => { event.preventDefault(); submit(); }}>
      <label htmlFor="composer" className="sr-only">提问</label>
      <textarea id="composer" data-testid="composer" rows={2} value={value}
        onChange={(event) => setValue(event.target.value)}
        onKeyDown={(event) => {
          if (event.key === "Enter" && !event.shiftKey && !event.nativeEvent.isComposing) {
            event.preventDefault();
            submit();
          }
        }}
        placeholder="用一句话描述症状，或输入方名 / 药名 / 穴位"
        className="w-full resize-y rounded border border-line bg-paper px-3 py-2 text-sm" />

        <span className="text-xs text-ink/50">Enter 发送，Shift+Enter 换行</span>
        {pending ? (
          <button type="button" onClick={onStop}
            className="rounded border border-cinnabar px-3 py-1 text-sm text-cinnabar">
            停止
          </button>
        ) : (
          <button type="submit" data-testid="send"
            className="rounded bg-cinnabar px-3 py-1 text-sm text-paper">
            发送
```

```tsx
function announcementFor(turn: Turn | undefined): string {
  switch (turn?.phase) {
    case "done": return "回答已完成";
    case "refused": return "已按证据门禁拒答";
    case "needs_more_info": return "需要补充问诊信息";
    case "error": return "生成失败";
    case "interrupted": return "回答已中断";
    default: return "";
  }
}

export function ChatPanel({
  initialSessionId = null,
  initialTurns = [],
  initialQuestion = "",
}: {
  initialSessionId?: number | null;
  initialTurns?: Turn[];
  initialQuestion?: string;
}) {
  const [state, dispatch] = useReducer(chatReducer, {
    ...initialChatState,
    sessionId: initialSessionId,
    turns: initialTurns,
  } satisfies ChatState);
  const [activeCitation, setActiveCitation] = useState<Citation | null>(null);
  const nextId = useRef(0);
  const { run, stop } = useChatTurn(dispatch);

  const sendQuestion = useCallback((question: string) => {
    if (state.pending) return;
    const id = `turn-${Date.now()}-${nextId.current++}`;
    dispatch({ type: "send", id, question });
    void run(id, question, state.sessionId);
  }, [run, state.pending, state.sessionId]);

  const lastTurn = state.turns[state.turns.length - 1];
  const announcement = announcementFor(lastTurn);

  return (
    <div className="mx-auto max-w-4xl px-4 py-4">
      <div data-testid="chat-transcript">
        {state.turns.map((turn) => (
          <div key={turn.id}>
            <MessageBubble role="user">{turn.question}</MessageBubble>
            <MessageBubble role="assistant">
              <div className="flex flex-wrap items-center gap-2">
                <IntentBadge intent={turn.intent} />
                {turn.phase === "sending" && (
                  <span data-testid="sending" className="text-xs text-ink/60">正在检索本地语料…</span>
                )}
              </div>
              {turn.progress && (
                <RetrievalProgress progress={turn.progress} citations={turn.citations} />
              )}
              {turn.phase === "streaming" && !turn.answer && (
                <p className="mt-2 whitespace-pre-wrap text-sm">{turn.streamText}</p>
              )}
              {turn.answer?.status === "refused" && (
                <RefusalCard answer={turn.answer} question={turn.question}
                  onAsk={sendQuestion} />
              )}
              {turn.answer?.status === "needs_more_info" && !turn.intakeSkipped && (
                <IntakeCard followup={turn.answer.followup} initialText={turn.question}
                  onSubmit={sendQuestion}
                  onSkip={() => dispatch({ type: "skip_intake", id: turn.id })} />
              )}
              {turn.intakeSkipped && (
                <p data-testid="intake-skip-note"
                  className="mt-2 rounded border border-dashed border-line bg-ink/5 p-2 text-xs text-ink/70">
                  基于不完整信息，以下辨证仅供参考
                </p>
              )}
              {turn.answer && turn.answer.status === "answered" && (
                <AnswerBody answer={turn.answer} onOpenCitation={setActiveCitation} />
              )}
              {turn.phase === "error" && (
                <div className="mt-2 flex items-center gap-2 text-sm text-cinnabar">

        onSend={sendQuestion} onStop={stop} />
      <div role="status" aria-live="polite" className="sr-only">{announcement}</div>
      <EvidenceDrawer citation={activeCitation} onClose={() => setActiveCitation(null)} />
    </div>
  );
}
```

```ts
// frontend/lib/history.ts
import type { Turn } from "./chatReducer";
import type { Answer, ChatHistoryMessage } from "./types";

export function historyToTurns(messages: ChatHistoryMessage[]): Turn[] {
  const turns: Turn[] = [];
  messages.forEach((message, index) => {
    if (message.role !== "user") return;
    const next = messages[index + 1];
    const answer = next?.role === "assistant"
      ? (next.content as unknown as Answer)
      : null;
    const question = typeof message.content.text === "string" ? message.content.text : "";
    turns.push({
      id: `history-${index}`,
      question,
      phase: answer
        ? answer.status === "refused" ? "refused"
          : answer.status === "needs_more_info" ? "needs_more_info" : "done"
        : "interrupted",
      sessionId: null,
      messageId: null,
      intent: answer?.intent ?? "",
      progress: null,
      citations: answer?.citations ?? [],
      streamText: "",
      answer,
      error: null,
      intakeSkipped: false,
    });
  });
  return turns;
}
```

```ts
import type { EvidenceRef } from "./types";

export function prefillFromEvidence(ref: EvidenceRef): string {
  const formula = ref.fields.formula as { canonical_name?: string } | undefined;
  const name = formula?.canonical_name
    ?? (typeof ref.fields.name === "string" ? ref.fields.name : undefined)
    ?? (typeof ref.fields.no === "string" ? ref.fields.no : undefined)
    ?? "";
  const evidence = Array.isArray(ref.fields.evidence)
    ? (ref.fields.evidence as {
        source_doc: string; six_jing: string | null; clause_no: string | null;
      }[])
    : [];
  if (evidence[0]) return prefillFromFormula(name, evidence);
  return `关于 ${name}（${ref.anchor}）：`;
}
```

```ts
export function paramsFrom(
  raw: Record<string, string | string[] | undefined>,
): URLSearchParams {
  const params = new URLSearchParams();
  for (const [key, value] of Object.entries(raw)) {
    if (Array.isArray(value)) {
      for (const item of value) if (item) params.append(key, item);
    } else if (value) {
      params.set(key, value);
    }
  }
  return params;
}
```

```tsx
"use client";

import Link from "next/link";
import { useEffect, useState } from "react";

import { search } from "@/lib/api";
import { refHref } from "@/lib/refs";
import type { SearchHit } from "@/lib/types";

const GROUPS = [
  ["exact", "精确"],
  ["literal", "字面"],
  ["vector", "语义相关"],
] as const;

export function SearchSuggest({
  kinds,
  initialQuery = "",
  onCommit,
}: {
  kinds: string[];
  initialQuery?: string;
  onCommit: (query: string) => void;
}) {
  const [query, setQuery] = useState(initialQuery);
  const [hits, setHits] = useState<SearchHit[]>([]);
  const [open, setOpen] = useState(false);

  useEffect(() => {
    const text = query.trim();
    if (!text) {
      setHits([]);
      setOpen(false);
      return;
    }
    let cancelled = false;
    const timer = setTimeout(() => {
      search(text, kinds)
        .then((response) => {
          if (cancelled) return;
          setHits(response.hits);
          setOpen(true);
        })
        .catch(() => {
          if (cancelled) return;
          setHits([]);
          setOpen(false);
        });
    }, 250);
    return () => { cancelled = true; clearTimeout(timer); };
  }, [query, kinds]);

  return (
    <div className="relative">
      <form role="search" onSubmit={(event) => {
        event.preventDefault();
        onCommit(query.trim());
        setOpen(false);
      }}>
        <label htmlFor="library-search" className="font-semibold">检索</label>
        <input id="library-search" type="search" aria-label="检索" value={query}
          onChange={(event) => setQuery(event.target.value)}
          onKeyDown={(event) => { if (event.key === "Escape") setOpen(false); }}
          placeholder="方名 / 药名 / 穴位 / 症状"
          className="mt-1 w-full rounded border border-line bg-paper px-2 py-1" />
      </form>
      {open && hits.length > 0 && (
        <div data-testid="suggest"
          className="absolute z-20 mt-1 w-full rounded border border-line bg-paper p-2 shadow">
          {GROUPS.map(([match, label]) => {
            const group = hits.filter((hit) => hit.match === match);
            if (!group.length) return null;
            return (
              <div key={match} data-testid="suggest-group">
                <p className="text-xs text-ink/60">{label}</p>
                <ul>
                  {group.map((hit) => {
                    const href = refHref(hit.ref_type, hit.ref_id);
                    return (
                      <li key={`${hit.ref_type}:${hit.ref_id}`}>
                        {href ? (
                          <Link href={href} onClick={() => setOpen(false)}
                            className="text-sm hover:text-cinnabar">{hit.title}</Link>
                        ) : <span className="text-sm">{hit.title}</span>}
                      </li>
                    );
                  })}
                </ul>
              </div>
            );
          })}
        </div>
      )}

  );
}
```

```tsx
"use client";

import { useRouter } from "next/navigation";
import { useEffect, useState } from "react";

import { buildQuery, type FilterValue } from "@/lib/filters";

import { SearchSuggest } from "./SearchSuggest";

export type FilterGroup =
  | { kind: "suggest"; key: string; label: string; kinds: string[] }
  | { kind: "multi"; key: string; label: string; options: string[] }
  | { kind: "single"; key: string; label: string; options: { value: string; label: string }[] }
  | { kind: "text"; key: string; label: string; placeholder?: string };

function TextFilter({ label, placeholder, value, onCommit }: {
  label: string;
  placeholder?: string;
  value: string;
  onCommit: (value: string) => void;
}) {
  const [text, setText] = useState(value);

  useEffect(() => {
    if (text === value) return;
    const timer = setTimeout(() => onCommit(text), 250);
    return () => clearTimeout(timer);
  }, [text, value, onCommit]);

  return (
    <label className="block">
      <span className="font-semibold">{label}</span>
      <input type="search" aria-label={label} value={text} placeholder={placeholder}
        onChange={(event) => setText(event.target.value)}
        className="mt-1 w-full rounded border border-line bg-paper px-2 py-1" />
    </label>
  );
}

export function FilterPanel({
  basePath,
  groups,
  initial,
}: {
  basePath: string;
  groups: FilterGroup[];
  initial: Record<string, string[]>;
}) {
  const router = useRouter();
  const [copied, setCopied] = useState(false);

  const push = (patch: Record<string, FilterValue>) => {
    router.push(`${basePath}${buildQuery({ ...initial, ...patch, page: null })}`);
  };

  async function copyLink() {
    await navigator.clipboard.writeText(window.location.href);
    setCopied(true);
    setTimeout(() => setCopied(false), 1500);
  }

  return (
    <aside aria-label="筛选" className="space-y-4 text-sm">
      {groups.map((group) => {
        if (group.kind === "suggest") {
          return (
            <SearchSuggest key={group.key} kinds={group.kinds}
              initialQuery={initial[group.key]?.[0] ?? ""}
              onCommit={(query) => push({ [group.key]: query || null })} />
          );
        }
        if (group.kind === "text") {
          return (
            <TextFilter key={group.key} label={group.label} placeholder={group.placeholder}
              value={initial[group.key]?.[0] ?? ""}
              onCommit={(value) => push({ [group.key]: value || null })} />
          );
        }
        if (group.kind === "multi") {
          const values = initial[group.key] ?? [];
          return (
            <fieldset key={group.key}>
              <legend className="font-semibold">{group.label}</legend>
              <div className="mt-1 space-y-1">
                {group.options.map((option) => {
                  const checked = values.includes(option);
                  return (
                    <label key={option} className="flex items-center gap-2">
                      <input type="checkbox" checked={checked}
                        onChange={() => push({
                          [group.key]: checked
                            ? values.filter((value) => value !== option)
                            : [...values, option],
                        })} />
                      {option}
                    </label>
                  );
                })}
              </div>
            </fieldset>
          );
        }
        return (
          <label key={group.key} className="block">
            <span className="font-semibold">{group.label}</span>
            <select aria-label={group.label} value={initial[group.key]?.[0] ?? ""}
              onChange={(event) => push({ [group.key]: event.target.value || null })}
              className="mt-1 w-full rounded border border-line bg-paper px-2 py-1">
              <option value="">全部</option>
              {group.options.map((option) => (
                <option key={option.value} value={option.value}>{option.label}</option>
              ))}
            </select>
          </label>
        );
      })}
      <button type="button" onClick={copyLink}
        className="rounded border border-line px-2 py-1 text-xs">
        {copied ? "已复制筛选链接" : "复制当前筛选链接"}
      </button>
    </aside>
  );
}
```

```tsx
// frontend/components/library/LibraryShell.tsx
import Link from "next/link";
import type { ReactNode } from "react";

import { PAGE_SIZE } from "@/lib/constants";
import { pageHref } from "@/lib/filters";

import { FilterPanel, type FilterGroup } from "./FilterPanel";

export function LibraryShell({
  title, total, basePath, params, groups, children,
}: {
  title: string;
  total: number;
  basePath: string;
  params: URLSearchParams;
  groups: FilterGroup[];
  children: ReactNode;
}) {
  const page = Number(params.get("page") ?? "1") || 1;
  const pages = Math.max(1, Math.ceil(total / PAGE_SIZE));
  const initial: Record<string, string[]> = {};
  params.forEach((value, key) => {
    if (key === "page") return;
    (initial[key] ??= []).push(value);
  });

  return (
    <div className="mx-auto grid max-w-7xl gap-6 px-4 py-4 md:grid-cols-[240px_1fr]">
      <FilterPanel basePath={basePath} groups={groups} initial={initial} />
      <section>
        <h1 className="font-serif text-2xl">{title}</h1>
        <p className="mt-1 text-sm text-ink/60">共 {total} 条</p>
        {total === 0 ? (
          <p className="mt-4 text-sm text-ink/60">
            没有匹配结果，换个筛选，或者
            <Link href="/chat" className="ml-1 text-cinnabar underline">去对话页问</Link>
          </p>
        ) : (
          <ul className="mt-2">{children}</ul>
        )}
        {pages > 1 && (
          <nav aria-label="分页" className="mt-4 flex items-center gap-3 text-sm">
            {page > 1 && <Link href={pageHref(basePath, params, page - 1)}>上一页</Link>}
            <span>第 {page} / {pages} 页</span>
            {page < pages && <Link href={pageHref(basePath, params, page + 1)}>下一页</Link>}
          </nav>
        )}
      </section>
    </div>
  );
}
```

```ts
import { notFound } from "next/navigation";

import { ApiError, serverGet } from "./api";

export function parseId(raw: string): number {
  const id = Number(raw);
  if (!Number.isInteger(id) || id <= 0) notFound();
  return id;
}

export async function loadDetail<T>(path: string): Promise<T> {
  try {
    return await serverGet<T>(path);
  } catch (error) {
    if (error instanceof ApiError && error.status === 404) notFound();
    throw error;
  }
}
```

```tsx
// frontend/components/chat/AnswerActions.tsx
"use client";

import { useState } from "react";

import { answerToMarkdown } from "@/lib/export";
import type { Answer } from "@/lib/types";

export function AnswerActions({ answer }: { answer: Answer }) {
  const [status, setStatus] = useState("");

  async function copy() {
    try {
      await navigator.clipboard.writeText(answerToMarkdown(answer));
      setStatus("已复制到剪贴板");
    } catch {
      setStatus("复制失败，请手动选择正文");
    }
  }

  function download() {
    const markdown = answerToMarkdown(answer);
    const blob = new Blob([markdown], { type: "text/markdown;charset=utf-8" });
    const url = URL.createObjectURL(blob);
    const anchor = document.createElement("a");
    anchor.href = url;
    anchor.download = `${(answer.headline || "answer").replace(/[\\/:*?"<>|]/g, "_")}.md`;
    anchor.click();
    URL.revokeObjectURL(url);
    setStatus("已导出 Markdown");
  }

  return (
    <div data-testid="answer-actions" className="mt-3 flex flex-wrap items-center gap-3 text-sm">
      <button type="button" onClick={copy} className="text-cinnabar underline">复制全文</button>
      <button type="button" onClick={download} className="text-cinnabar underline">
        导出 Markdown
      </button>
      <span role="status" aria-live="polite" className="text-xs text-ink/70">{status}</span>
    </div>
  );
}
```
